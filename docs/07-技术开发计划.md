### 4.3 进度管理（LocalStorage + 导出/导入）

```javascript
const STORAGE_KEY = 'guanzhi_progress';
const PROGRESS_VERSION = 1;

function getProgress() {
  const data = localStorage.getItem(STORAGE_KEY);
  return data ? JSON.parse(data) : {
    version: PROGRESS_VERSION,
    studentName: '',
    completedArticles: [],
    completedWeeks: [],
    quizScores: {},
    reviewPassed: [],
    badges: [],
    vocabLearned: 0,
    lastVisit: new Date().toISOString().split('T')[0],
    currentWeek: 1,
    createdAt: new Date().toISOString().split('T')[0],
    exportedAt: null
  };
}

function saveProgress(progress) {
  progress.lastVisit = new Date().toISOString().split('T')[0];
  localStorage.setItem(STORAGE_KEY, JSON.stringify(progress));
}

// === 导出进度为 JSON 文件 ===
function exportProgress() {
  const progress = getProgress();
  progress.exportedAt = new Date().toISOString();
  const name = progress.studentName || '学生';
  const date = new Date().toISOString().split('T')[0];
  const filename = `${name}-进度-${date}.json`;
  const blob = new Blob([JSON.stringify(progress, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}

// === 从 JSON 文件导入进度 ===
function importProgress(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = JSON.parse(e.target.result);
        // 数据校验
        if (!data.version || !Array.isArray(data.completedArticles)) {
          reject(new Error('文件格式不正确'));
          return;
        }
        if (data.version > PROGRESS_VERSION) {
          reject(new Error('进度文件版本较新，请更新网站后再导入'));
          return;
        }
        saveProgress(data);
        resolve(data);
      } catch (err) {
        reject(new Error('无法解析文件内容'));
      }
    };
    reader.readAsText(file);
  });
}

// === 备份提醒（每隔2周检查一次） ===
function checkBackupReminder() {
  const progress = getProgress();
  const lastExport = progress.exportedAt 
    ? new Date(progress.exportedAt) 
    : new Date(progress.createdAt);
  const daysSinceExport = (Date.now() - lastExport.getTime()) / (1000 * 60 * 60 * 24);
  if (daysSinceExport >= 14 && progress.completedArticles.length > 0) {
    return true; // 应弹出备份提醒
  }
  return false;
}
```

### 4.4 原文拼音渲染

```javascript
// JSON中拼音数据格式
{
  "text": "曹刿请见",
  "ruby": [
    { "char": "刿", "pinyin": "guì", "pos": 1 }
  ]
}

// 渲染函数
function renderTextWithRuby(text, rubyData) {
  let html = '';
  for (let i = 0; i < text.length; i++) {
    const ruby = rubyData.find(r => r.pos === i);
    if (ruby) {
      html += `<ruby>${text[i]}<rp>(</rp><rt>${ruby.pinyin}</rt><rp>)</rp></ruby>`;
    } else {
      html += text[i];
    }
  }
  return html;
}
```

### 4.5 字词注释弹窗

```javascript
// 给原文中的注释词汇添加可点击标记
function renderAnnotatedText(text, annotations) {
  let html = text;
  // 按长度降序排列，避免短词覆盖长词
  annotations.sort((a, b) => b.word.length - a.word.length);
  annotations.forEach(ann => {
    html = html.replace(
      ann.word,
      `<span class="annotated" data-meaning="${ann.meaning}" 
       data-pinyin="${ann.pinyin || ''}">${ann.word}</span>`
    );
  });
  return html;
}

// 点击/hover显示弹窗
document.addEventListener('click', (e) => {
  if (e.target.classList.contains('annotated')) {
    showTooltip(e.target, {
      word: e.target.textContent,
      meaning: e.target.dataset.meaning,
      pinyin: e.target.dataset.pinyin
    });
  }
});
```

function handleChoiceClick(questionId, selectedOption, correctAnswer) {
  const options = document.querySelectorAll(`#q${questionId} .option`);
  options.forEach(opt => opt.style.pointerEvents = 'none'); // 禁止重复作答
  
  const selected = document.querySelector(`#q${questionId}-${selectedOption}`);

## 八、给 Codex 的开发指引

### 关键注意事项

1. **不使用任何框架**：纯 HTML/CSS/JS，不引入 React/Vue/Angular
2. **中文内容处理**：确保 UTF-8 编码，字体渲染清晰
3. **移动端优先**：学生可能用手机或平板学习
4. **无需后端**：所有数据通过 JSON 文件 + LocalStorage 管理
5. **图片格式统一为 JPG**：放在 `images/` 目录下
